```python
class AnalysisOrchestrator:
    """LLM-driven iterative analysis loop (D-01).
    
    Single LLM orchestrator per iteration decides which stage to run next.
    Stages: issue-spot → research → fact-map → gap-analyze → question.
    """
    
    STAGES = ["issue_spot", "research", "fact_map", "gap_analyze", "question_gen"]
    
    def __init__(self, db_session, llm_service, folio, embedding_service, org_config=None):
        # Store services, initialize stage instances
    
    async def run(self, intake_id: int, session_id: int, ws_manager=None) -> AnalysisRun:
        """Execute the full analysis loop.
        
        1. Create AnalysisRun record
        2. Loop: run iteration until convergence or hard cap
        3. On each iteration:
           a. Create AnalysisIteration record
           b. LLM decides which stages to run (can skip irrelevant ones)
           c. Execute each selected stage, creating AnalysisStage checkpoint after each
           d. Evaluate convergence after gap analysis
           e. Send WebSocket progress update via ws_manager.send_to_session()
        4. On convergence: create final summary, update AnalysisRun status
        5. Return AnalysisRun with final state
        """
    
    async def resume(self, run_id: int, ws_manager=None) -> AnalysisRun:
        """Resume from latest checkpoint (ANALYSIS-09).
        Load latest AnalysisStage, continue from next stage in sequence.
        """
    
    async def override_convergence(self, run_id: int) -> AnalysisRun:
        """Override termination and continue analysis (D-16).
        Reset convergence signals, increment iteration, continue loop.
        """
    
    async def _run_iteration(self, run, iteration_num, ws_manager) -> AnalysisIteration:
        """Single iteration: LLM selects stages → execute → checkpoint → converge check."""
    
    async def _select_stages(self, run, iteration) -> list[str]:
        """LLM orchestrator decides which stages to run this iteration."""
    
    async def _execute_stage(self, stage_name, run, iteration, ws_manager):
        """Execute a single stage, create AnalysisStage checkpoint, broadcast progress."""
    
    async def _broadcast_progress(self, run, stage_name, ws_manager):
        """Push stage-by-stage progress update via WebSocket (D-15)."""
```

```python
class AnalysisTrigger:
    """Auto + manual analysis triggering (D-04).
    
    Auto-triggers when N new facts accumulate since last analysis run.
    Manual trigger available via REST API at any time.
    Threshold configurable per org via analysis_settings.auto_trigger_threshold.
    """
    
    def __init__(self, db_session, orchestrator: AnalysisOrchestrator):
        pass
    
    async def check_auto_trigger(self, intake_id: int, session_id: int) -> bool:
        """Check if auto-trigger threshold reached. Called after fact extraction.
        Count new ExtractedFact records since last AnalysisRun for this intake.
        If count >= threshold and no analysis currently running: trigger.
        Returns True if analysis was triggered.
        """
    
    async def manual_trigger(self, intake_id: int, session_id: int) -> AnalysisRun:
        """Manually trigger analysis. Always runs regardless of fact count.
        If analysis already running for this intake: return existing run.
        """
    
    async def _is_analysis_running(self, intake_id: int) -> bool:
        """Check if an AnalysisRun with status='running' exists for this intake."""
```

```python
router = APIRouter(prefix="/api/v1/analysis", tags=["analysis"])

@router.post("/{intake_id}/analyze", status_code=202)
async def trigger_analysis(intake_id: int, ...):
    """Manual analysis trigger. Returns 202 Accepted with run_id.
    Analysis runs as background asyncio.create_task (D-02 hybrid model).
    """

@router.get("/{intake_id}/status")
async def get_analysis_status(intake_id: int, ...):
    """Get current analysis run status: iteration, stage, convergence score, progress %."""

@router.get("/{intake_id}/results")  
async def get_analysis_results(intake_id: int, ...):
    """Get analysis results: claims, mappings, gaps, questions. Includes jurisdiction labels."""

@router.post("/{intake_id}/override")
async def override_convergence(intake_id: int, ...):
    """Override convergence and continue analysis (D-16). Consumer or professional."""

@router.get("/{intake_id}/audit")
async def get_audit_trail(intake_id: int, ...):
    """Get full audit trail: every stage, sources consulted, confidence scores (ANALYSIS-10)."""
```
