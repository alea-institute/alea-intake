    class TriageScorer:
        """Multi-factor triage and routing scorer per D-02."""

        Configurable default weights:
        PRACTICE_AREA_WEIGHT = 0.35
        JURISDICTION_WEIGHT = 0.25
        COMPLEXITY_WEIGHT = 0.20
        ORG_RULES_WEIGHT = 0.20

        def score(self, context: OutputContext, org_routing_rules: dict | None = None) -> TriageResult:
            1. Extract distinct practice areas from claims (derive from claim_type, folio_iri)
            2. Extract distinct jurisdictions from claims
            3. Compute complexity_level from claim count + gap count
            4. Compute urgency_level from max gap priority
            5. For each practice area, build TriageRecommendation:
               - practice_area_match = proportion of claims in this area
               - jurisdiction_match = 1.0 if single jurisdiction, fraction if mixed
               - complexity_score = inverse complexity (simple cases easier to route)
               - score = weighted sum of factors
               - rationale = human-readable explanation
            6. If org_routing_rules provided, apply org-specific adjustments (boost/penalize destinations)
            7. Sort recommendations by score desc
            8. Return TriageResult with primary_practice_area (top recommendation), primary_jurisdiction (most common)

    class ActionItemGenerator:
        """Transforms gaps into prioritized, categorized action items per D-03."""

        GAP_TYPE_TO_CATEGORY mapping:
        - unsupported_element -> documents_to_gather
        - weak_mapping -> documents_to_gather
        - unexplored_claim -> follow_up_steps
        - procedural_requirement -> follow_up_steps

        PRIORITY_THRESHOLDS:
        - >= 8 -> "urgent"
        - >= 5 -> "important"
        - else -> "helpful"

        def generate(
            self,
            gap_report: GapReport,
            claims_by_jurisdiction: dict[str, list[CIRACSection]],
        ) -> list[ActionItem]:
            1. Iterate consolidated_gaps, create ActionItem for each
            2. Assign category from gap_type mapping
            3. Assign priority from gap priority value
            4. Set claim_ref and element_ref from gap.claim_name and gap.element_name
            5. Set deadline from gap metadata if procedural_requirement
            6. Number items sequentially (1, 2, 3...)
            7. Cross-link: set gap_report entry's action_item_ref to the item_number
            8. Sort: urgent first, then important, then helpful; within same priority, documents_to_gather before follow_up_steps before referrals
            9. Referral items: create for practice areas that have completeness < 0.3 (severe gaps suggesting outside referral)
            Return sorted list[ActionItem]

    class TemplateEngine:
        """Renders OutputContext through Jinja2 templates per profile (per research Pattern 1)."""

        DEFAULT_TEMPLATE_DIR = Path(__file__).parent / "templates"

        def __init__(self, template_dir: str | Path | None = None):
            self._env = Environment(
                loader=FileSystemLoader(str(template_dir or self.DEFAULT_TEMPLATE_DIR)),
                autoescape=False,  # Markdown output, not HTML
                trim_blocks=True,
                lstrip_blocks=True,
            )
            Register custom filters: datetimeformat, truncate, percentage (float -> "XX%")

        def render_full(self, context: OutputContext, profile: OutputProfile) -> str:
            Concatenate rendered sections based on profile.sections visibility:
            - If executive_summary and sections["executive_summary"]: render exec summary
            - If sections["cirac_memo"]: render cirac_memo.md.j2
            - If sections["triage_routing"]: render triage_report.md.j2
            - If sections["action_items"]: render action_items.md.j2
            - If sections["gap_appendix"]: render gap_appendix.md.j2
            Return concatenated Markdown string

        def render_section(self, template_name: str, context: OutputContext, profile: OutputProfile) -> str:
            Render single section template.

    class LanguageAdapter:
        """LLM-driven language complexity adaptation per profile (per D-05)."""

        SYSTEM_PROMPTS by language_level:
        - "professional": None (skip LLM, use text as-is)
        - "accessible": "Rewrite the following legal text to be accessible to a non-lawyer. Keep all citation strings (e.g., '123 F.3d 456') exactly as-is. Keep all legal terms of art but add brief parenthetical explanations. Target reading level: 10th grade."
        - "plain": "Rewrite the following legal text in plain language at an 8th grade reading level. Keep all citation strings exactly as-is. Replace legal jargon with everyday language. Add brief explanations for necessary legal terms."

        async def adapt(self, context: OutputContext, profile: OutputProfile, llm_service: LLMService) -> OutputContext:
            If language_level == "professional": return context unchanged
            Otherwise:
            1. Extract all citation strings from context (from authorities) to preserve
            2. Rewrite executive_summary via LLM
            3. For each CIRACSection: rewrite issue_statement and conclusion via LLM
               (Do NOT rewrite authority citations, element names, or fact text -- these are data, not prose)
            4. Return new OutputContext with rewritten text fields
            Use alea-llm-client via LLMService pattern from existing code.
            IMPORTANT per pitfall 3: Post-process to verify all citation strings survive in output. If any dropped, restore from original.
