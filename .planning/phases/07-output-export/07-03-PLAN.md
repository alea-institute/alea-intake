    Create backend/app/services/output/export/pdf_adapter.py:
    class PDFAdapter(ExportAdapter):
        CSS_PATH = Path(__file__).parent.parent / "templates" / "css" / "legal_pdf.css"

        async def export(self, markdown: str, context: OutputContext, profile: OutputProfile) -> bytes:
            1. Convert Markdown to HTML via markdown-it-py: MarkdownIt("commonmark", {"breaks": True, "html": True}).enable("table").render(markdown)
            2. Wrap in full HTML document with <html><head><meta charset="utf-8"><style>{css}</style></head><body>{html}</body></html>
            3. Inject org_name and matter_title as hidden elements for CSS string-set (running headers/footers)
            4. If profile.org_branding: inject custom CSS variables for colors
            5. Render via WeasyPrint: HTML(string=full_html).write_pdf()
            6. Return PDF bytes
        content_type = "application/pdf"
        file_extension = "pdf"

    Create backend/app/services/output/templates/css/legal_pdf.css:
    Use the CSS from research (Pattern 4: CSS Paged Media for Legal PDF) as base:
    - @page: letter size, 1in margins (1.25in bottom), top-center = matter title, bottom-center = page counter, bottom-right = org name
    - @page :first: suppress header/footer on first page
    - Typography: body font 12pt, serif (Times New Roman equivalent)
    - Numbered paragraphs via CSS counter (.numbered-para)
    - H1 centered, H2 with top border, H3 with left border
    - Authority formatting: binding = bold, persuasive = italic, secondary = normal
    - Verified/unverified badges
    - Table styling for any data tables
    - Page break hints: avoid break inside claim sections, break before new jurisdiction
    - Print-friendly colors (no bright backgrounds)
    - CSS variables for org branding: --primary-color, --secondary-color

    Create backend/app/services/output/export/docx_adapter.py:
    class DOCXAdapter(ExportAdapter):
        async def export(self, markdown: str, context: OutputContext, profile: OutputProfile) -> bytes:
            1. Create Document()
            2. Set default style: Times New Roman 12pt
            3. Add header with matter_title, footer with "Confidential - Attorney Work Product"
            4. Add TOC placeholder via add_toc_placeholder() (use the XML workaround from research)
            5. Parse Markdown line by line:
               - Lines starting with # -> doc.add_heading(text, level=N)
               - Lines starting with - -> doc.add_paragraph(text, style="List Bullet")
               - Lines starting with \d+. -> doc.add_paragraph(text, style="List Number")
               - **bold** -> run with bold=True
               - *italic* -> run with italic=True
               - Plain text -> doc.add_paragraph(text)
               - --- -> page break
            6. Save to BytesIO, return bytes
        content_type = "application/vnd.openxmlformats-officedocument.wordprocessingml.document"
        file_extension = "docx"

        Helper: add_toc_placeholder(doc) -- insert TOC field code per research example.

    Pydantic request/response schemas (inline or in separate schemas file):
    - GenerateRequest: run_id (int), intake_id (int), profile_types (list[str] | None, default None)
    - OutputDocumentSummary: id, run_id, intake_id, profile_type, completeness_score, created_at
    - OutputDocumentDetail: all of summary + markdown_content (truncated in list, full in detail)
    - GenerateResponse: documents (list[OutputDocumentSummary])

    Endpoints:

    @router.post("/generate", response_model=GenerateResponse, status_code=201)
    async def generate_output(
        request: GenerateRequest,
        db: AsyncSession = Depends(get_db_session),
        current_user = Depends(require_auth),
    ):
        1. Determine profiles:
           - If request.profile_types provided: resolve each to OutputProfile (LAW_FIRM_PROFILE, etc.)
           - Else: load org's output_config_json from OrganizationConfig, default to LAW_FIRM_PROFILE
        2. Create DataAssembler(db)
        3. For each profile:
           a. assembler.assemble(run_id, intake_id, profile) -> context
           b. TriageScorer().score(context) -> triage (set on context)
           c. ActionItemGenerator().generate(context.gap_report, context.claims_by_jurisdiction) -> action_items (set on context)
           d. LanguageAdapter().adapt(context, profile, get_llm_service()) -> adapted_context (only if non-professional)
           e. TemplateEngine().render_full(adapted_context, profile) -> markdown
           f. Create OutputDocument(run_id, intake_id, profile_type, markdown_content=markdown)
           g. db.add(doc), flush to get ID
        4. await db.commit()
        5. Return list of OutputDocumentSummary

    @router.get("/intake/{intake_id}", response_model=list[OutputDocumentSummary])
    async def list_outputs(intake_id: int, db, current_user):
        SELECT from output_documents WHERE intake_id=? ORDER BY created_at DESC

    @router.get("/{document_id}", response_model=OutputDocumentDetail)
    async def get_output(document_id: int, db, current_user):
        SELECT from output_documents WHERE id=?
        Return full detail with markdown_content

    @router.get("/{document_id}/export/{format}")
    async def export_output(document_id: int, format: str, db, current_user):
        1. Load OutputDocument from DB
        2. Select adapter: {"pdf": PDFAdapter(), "docx": DOCXAdapter(), "json": JSONAdapter()}
        3. If rendered bytes already cached on document (rendered_pdf/rendered_docx/rendered_json): return cached
        4. Else: render via adapter.export(doc.markdown_content, context, profile)
        5. Cache rendered bytes on document model, commit
        6. Return StreamingResponse with adapter.content_type and Content-Disposition attachment header
