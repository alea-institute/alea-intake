    class DataAssembler:
        """Queries all analysis/research data into a unified OutputContext (per research Pattern 1)."""

        def __init__(self, db_session: AsyncSession):
            self._session = db_session

        async def assemble(self, run_id: int, intake_id: int, profile: OutputProfile) -> OutputContext:
            Load all data in parallel where possible:
            1. _load_claims(run_id) -- SELECT from analysis_claims WHERE run_id=? ORDER BY jurisdiction, claim_name
            2. _load_elements(claim_ids) -- SELECT from claim_elements WHERE claim_id IN (?)
            3. _load_mappings(run_id, fact_ids) -- SELECT from fact_claim_mappings WHERE claim_id IN (?) -- join with extracted_facts for assertion_text
            4. _load_gaps(run_id) -- SELECT from analysis_gaps WHERE run_id=? AND status='open'
            5. _load_questions(run_id) -- SELECT from follow_up_questions WHERE run_id=?
            6. _load_authorities(intake_id) -- SELECT from authorities WHERE intake_id=?
            7. _load_facts(intake_id) -- SELECT from extracted_facts WHERE intake_id=? AND is_active=True
            8. _load_run(run_id) -- SELECT from analysis_runs WHERE id=?

            Build CIRACSection for each claim:
            - issue_statement = claim.rationale or f"Whether {claim.claim_name} applies based on the presented facts"
            - authorities = filter authorities by claim_iri matching claim.folio_iri, sorted by binding_strength priority (binding=0, persuasive=1, secondary=2) then relevance_score desc
            - elements = ClaimElement records for this claim, each with its FactMappingRef list
            - gaps = AnalysisGap records where claim_id matches
            - conclusion = derive from element satisfaction ratio: "X of Y elements supported (Z% confidence)"

            Group CIRACSection by jurisdiction (claims with None jurisdiction -> "General").

            Build GapReport via GapReportBuilder.build().

            Return OutputContext with:
            - matter_title from intake metadata_json.get("title", f"Intake #{intake_id}")
            - generated_at = datetime.utcnow()
            - completeness_score from run.convergence_score or compute from element satisfaction ratio
            - executive_summary = "" (placeholder for LLM adaptation in Plan 02)
            - profile = profile parameter

    class GapReportBuilder:
        """Builds inline + appendix gap analysis per D-07."""

        @staticmethod
        def build(
            gaps: list[AnalysisGap],
            questions: list[FollowUpQuestion],
            claims: dict[int, str],  # claim_id -> claim_name
            elements: dict[int, str],  # element_id -> element_name
            convergence_score: float | None,
        ) -> GapReport:
            Build per_claim dict: group gaps by claim_id, resolve claim_name from claims dict.
            Build consolidated_gaps: all gaps sorted by priority desc.
            Build open_questions: [q.question_text for q in questions if q.status == "pending"].
            completeness_score = convergence_score if convergence_score is not None else 0.0.
            Return GapReport.
