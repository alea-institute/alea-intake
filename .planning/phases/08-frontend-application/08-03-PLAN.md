<!-- User upsert contract -->

SSOService.upsert_user(email, provider, provider_id, full_name?, org_id?):
  - If user exists by email: link SSO provider (update sso_provider, sso_subject fields)
  - If user doesn't exist: create with role=consumer (default), org_id=provided-or-default
  - Returns User model

SSOService.generate_exchange_code(user_id, access_token):
  - Creates short-lived (60s) one-time nonce
  - Stored in Redis or in-memory cache keyed by nonce → (user_id, access_token)
  - Returns nonce string

SSOService.redeem_exchange_code(nonce):
  - Looks up nonce, invalidates, returns (access_token, User)
  - Raises 400 if expired/invalid

<!-- Frontend OAuth flow -->

1. User clicks SSO button on /login → window.location = /api/v1/auth/oauth/login/google
2. Backend 302 → Google consent
3. Google 302 → /api/v1/auth/oauth/callback/google?code=...&state=...

7. Backend redeems nonce → returns { access_token, user } in JSON

    ```python
    import secrets
    import time
    from dataclasses import dataclass
    from typing import Optional
    from sqlalchemy.ext.asyncio import AsyncSession
    from sqlalchemy import select
    from app.models.user import User, Role

    @dataclass
    class _NonceEntry:
        user_id: str
        access_token: str
        expires_at: float

    class SSOService:
        """OAuth user upsert + one-time-code exchange (Pitfall 4 safe pattern)."""

        # In-memory nonce store (MVP; Redis in production for multi-worker)
        _nonces: dict[str, _NonceEntry] = {}
        _NONCE_TTL_SEC = 60

        def __init__(self, session: AsyncSession):
            self._session = session

        async def upsert_user(
            self,
            *,
            email: str,
            provider: str,
            provider_id: str,
            full_name: Optional[str] = None,
            org_id: Optional[str] = None,
        ) -> User:
            # Try by (provider, subject) first — strongest identity
            stmt = select(User).where(User.sso_provider == provider, User.sso_subject == provider_id)
            result = await self._session.execute(stmt)
            user = result.scalar_one_or_none()

            if user:
                return user

            # Fallback: by email (link existing account)
            stmt = select(User).where(User.email == email)
            result = await self._session.execute(stmt)
            user = result.scalar_one_or_none()

            if user:
                user.sso_provider = provider
                user.sso_subject = provider_id
                if full_name and not user.full_name:
                    user.full_name = full_name
                await self._session.flush()
                return user

            # Create new user
            if not org_id:
                raise ValueError("org_id required for new SSO user creation")
            user = User(
                email=email,
                full_name=full_name,
                role=Role.CONSUMER,
                org_id=org_id,
                sso_provider=provider,
                sso_subject=provider_id,
                password_hash=None,  # SSO-only user
            )
            self._session.add(user)
            await self._session.flush()
            return user

        @classmethod
        def generate_exchange_code(cls, user_id: str, access_token: str) -> str:
            # Reap expired entries
            now = time.time()
            expired = [n for n, e in cls._nonces.items() if e.expires_at < now]
            for n in expired:
                cls._nonces.pop(n, None)

            nonce = secrets.token_urlsafe(32)
            cls._nonces[nonce] = _NonceEntry(
                user_id=user_id,
                access_token=access_token,
                expires_at=now + cls._NONCE_TTL_SEC,
            )
            return nonce

        @classmethod
        def redeem_exchange_code(cls, nonce: str) -> tuple[str, str]:
            entry = cls._nonces.pop(nonce, None)
            if not entry:
                raise ValueError("Invalid or already-used code")
            if entry.expires_at < time.time():
                raise ValueError("Code expired")
            return entry.user_id, entry.access_token
    ```

    **Step 1 — Create `backend/app/routers/oauth.py`:**

    ```python
    from fastapi import APIRouter, Request, HTTPException, Depends
    from fastapi.responses import RedirectResponse, JSONResponse
    from pydantic import BaseModel
    from sqlalchemy.ext.asyncio import AsyncSession
    from app.core.oauth import oauth
    from app.services.sso_service import SSOService
    from app.services.auth_service import AuthService
    from app.database import get_session
    from app.config import get_settings

    router = APIRouter(prefix="/auth/oauth", tags=["oauth"])
    settings = get_settings()

    SUPPORTED_PROVIDERS = {"google", "microsoft"}

    @router.get("/login/{provider}")
    async def oauth_login(provider: str, request: Request):
        if provider not in SUPPORTED_PROVIDERS:
            raise HTTPException(404, f"Unknown provider: {provider}")
        client = oauth.create_client(provider)
        if client is None:
            raise HTTPException(503, f"{provider} SSO not configured")
        redirect_uri = f"{settings.oauth_redirect_base_url}/api/v1/auth/oauth/callback/{provider}"
        return await client.authorize_redirect(request, redirect_uri)

    @router.get("/callback/{provider}")
    async def oauth_callback(
        provider: str,
        request: Request,
        session: AsyncSession = Depends(get_session),
    ):
        if provider not in SUPPORTED_PROVIDERS:
            raise HTTPException(404, f"Unknown provider: {provider}")
        client = oauth.create_client(provider)
        if client is None:
            raise HTTPException(503, f"{provider} SSO not configured")

        try:
            token = await client.authorize_access_token(request)
        except Exception as e:
            raise HTTPException(400, f"OAuth exchange failed: {e}")

        userinfo = token.get("userinfo")
        if not userinfo:
            userinfo = await client.parse_id_token(request, token)

        email = userinfo.get("email")
        if not email:
            raise HTTPException(400, "Provider did not return email")

        sso_service = SSOService(session)
        auth_service = AuthService(session)

        # Determine org: use request org context OR default org (MVP: first org)
        org_id = getattr(request.state, "org_id", None) or await auth_service.get_default_org_id()

        user = await sso_service.upsert_user(
            email=email,
            provider=provider,
            provider_id=userinfo.get("sub", ""),
            full_name=userinfo.get("name"),
            org_id=org_id,
        )
        await session.commit()

        access_token, refresh_token = await auth_service.mint_tokens_for_user(user)

        # Generate one-time exchange code (Pitfall 4 — no token in URL)
        nonce = SSOService.generate_exchange_code(str(user.id), access_token)

        response = RedirectResponse(
            url=f"{settings.frontend_base_url}/oauth/finish?code={nonce}",
            status_code=302,
        )
        response.set_cookie(
            "refresh_token",
            refresh_token,
            httponly=True,
            secure=(settings.environment == "production"),
            samesite="lax",
            max_age=7 * 24 * 3600,
            path="/api/v1/auth/refresh",
        )
        return response

    class ExchangeRequest(BaseModel):
        code: str

    @router.post("/exchange")
    async def oauth_exchange(
        body: ExchangeRequest,
        session: AsyncSession = Depends(get_session),
    ):
        try:
            user_id, access_token = SSOService.redeem_exchange_code(body.code)
        except ValueError as e:
            raise HTTPException(400, str(e))

        auth_service = AuthService(session)
        user = await auth_service.get_user_by_id(user_id)
        if not user:
            raise HTTPException(404, "User not found")

        return JSONResponse({
            "access_token": access_token,
            "user": {
                "id": str(user.id),
                "email": user.email,
                "role": user.role.value,
                "org_id": str(user.org_id),
                "full_name": user.full_name,
            },
    ```

    ```tsx
    import { useEffect, useState } from 'react'
    import { useNavigate, useSearchParams } from 'react-router-dom'
    import { useTranslation } from 'react-i18next'
    import { useAuth } from './store'

    export function OAuthFinishPage() {
      const { t } = useTranslation('auth')
      const [params] = useSearchParams()
      const navigate = useNavigate()
      const [error, setError] = useState('')

      useEffect(() => {
        const code = params.get('code')
        if (!code) { setError(t('errors.ssoMissingCode', 'Sign-in failed. Please try again.')); return }
        ;(async () => {
          try {
            const res = await fetch('/api/v1/auth/oauth/exchange', {
              method: 'POST',
              credentials: 'include',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify({ code }),
            })
            if (!res.ok) throw new Error('exchange failed')
            const { access_token, user } = await res.json()
            useAuth.getState().setAuth(access_token, user)
            navigate('/dashboard', { replace: true })
          } catch {
            setError(t('errors.ssoFailed', 'We couldn\\'t complete sign-in. Please try again.'))
          }
        })()
      }, [params, navigate, t])

      return (
        <div className="min-h-screen flex items-center justify-center p-lg">
          {error ? (
            <div className="text-destructive" role="alert">{error}</div>
          ) : (
            <div className="text-muted-foreground">{t('finishing', 'Completing sign-in…')}</div>
          )}
        </div>
      )
    }
    ```
