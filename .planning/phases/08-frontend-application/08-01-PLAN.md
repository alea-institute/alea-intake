    ```js
    #!/usr/bin/env node
    import { readdirSync, statSync, readFileSync } from 'fs'
    import { join } from 'path'
    import { gzipSync } from 'zlib'

    const DIST = 'dist/assets'
    const MAX_MAIN_GZIP_KB = 200

    function findMainChunk() {
      const files = readdirSync(DIST)
      // Main chunk is the entry JS (not a vendor chunk)
      return files.find(f => f.startsWith('index-') && f.endsWith('.js'))
    }

    const main = findMainChunk()
    if (!main) { console.error('No main chunk found in', DIST); process.exit(1) }

    const content = readFileSync(join(DIST, main))
    const gzipped = gzipSync(content)
    const kb = (gzipped.length / 1024).toFixed(2)

    console.log(`Main chunk: ${main}`)
    console.log(`  Raw: ${(content.length / 1024).toFixed(2)} KB`)
    console.log(`  Gzipped: ${kb} KB`)
    console.log(`  Budget: ${MAX_MAIN_GZIP_KB} KB`)

    if (gzipped.length / 1024 > MAX_MAIN_GZIP_KB) {
      console.error(`❌ Bundle exceeds budget by ${(kb - MAX_MAIN_GZIP_KB).toFixed(2)} KB`)
      process.exit(1)
    }
    console.log('✅ Bundle within budget')
    ```
